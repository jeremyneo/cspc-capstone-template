% Unnumbered layout
\newcommand{\unnumberedlayout}{%
    \newgeometry{
        top=1.35in,
        bottom=1in,
        left=1.5in,
        right=1in,
        headheight=0.9in,
        headsep=0.4in,
        footskip=0.5in
    }%
    \pagestyle{unnumbered}%
}

% Numbered layout
\newcommand{\numberedlayout}{%
    \newgeometry{
        top=1.55in,
        bottom=1in,
        left=1.5in,
        right=1in,
        headheight=1in,
        headsep=0.4in,
        footskip=0.5in
    }%
    \pagestyle{numbered}%
}

\newcommand{\preliminarylayout}{%
    \unnumberedlayout
    \pagestyle{preliminary}%
}

\newcommand{\chapterlayout}{%
    \numberedlayout
    \thispagestyle{unnumbered}%
    \doublespacing
}

\newcommand{\PreliminaryTitle}[2][true]{%
    \phantomsection
    \addcontentsline{toc}{section}{#2}%
    \ifstrequal{#1}{true}{%
        \centerBold{\MakeUppercase{#2}}%
    }{}%
}

\newcommand{\ChapterTitle}[1]{%
    \refstepcounter{section}%
    \phantomsection
    \addcontentsline{toc}{section}{CHAPTER \thesection: \MakeUppercase{#1}}%
    \begin{center}
        \textbf{CHAPTER \thesection}\par
        \textbf{\MakeUppercase{#1}}\par
    \end{center}%
}
